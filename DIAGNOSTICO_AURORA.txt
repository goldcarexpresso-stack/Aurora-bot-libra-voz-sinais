4. CODIGO ATUAL FUNCIONANDO (App.js Passo 4):
import { useState } from 'react';
import { Text, View, TouchableOpacity, StyleSheet, Platform } from 'react-native';

export default function App() {
  const [texto, setTexto] = useState('Oi! Eu sou a Aurora!');
  const [ouvindo, setOuvindo] = useState(false);
  const [emLibras, setEmLibras] = useState(false);
  const [glosa, setGlosa] = useState('');

  function traduzirParaLibras(frase) {
    let g = frase.toUpperCase();
    g = g.replace(/VOCES/g, 'VOCE').replace(/ESTA/g, 'ESTAR').replace(/TUDO BEM/g, 'TUDO BEM?');
    setGlosa(g);
    setEmLibras(true);
  }

  function comecarOuvir() {
    setOuvindo(true);
    setTexto('Ouvindo... pode falar!');
    setEmLibras(false);
    if (Platform.OS === 'web') {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (SpeechRecognition) {
        const rec = new SpeechRecognition();
        rec.lang = 'pt-BR';
        rec.onresult = (e) => {
          const t = e.results[0][0].transcript;
          setTexto(t);
          traduzirParaLibras(t);
          setOuvindo(false);
        };
        rec.onerror = () => setOuvindo(false);
        rec.onend = () => setOuvindo(false);
        rec.start();
      }
    }
  }

  function pararOuvir() {
    setOuvindo(false);
    if (Platform.OS!== 'web' &&!emLibras) {
      const t = 'Oi tudo bem que bom te ver';
      setTexto(t);
      traduzirParaLibras(t);
    }
  }
  // ... resto com 2 telas
}
